usage: ray-vast up [-h] [--profile PROFILE] [--wheel WHEEL] cluster_yaml

positional arguments:
  cluster_yaml

options:
  -h, --help         show this help message and exit
  --profile PROFILE  AWS profile (default: AWS_PROFILE; else, for down, attach
                     and dashboard, the profile that up saved; else the
                     default credentials)
  --wheel WHEEL      ray-vast wheel to install on the head, for development
                     (default: in a source tree, build the wheel; else install
                     this CLI's own version from PyPI)
